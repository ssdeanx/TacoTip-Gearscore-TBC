local function findInspector()
    local libStub = _G.LibStub
    if type(libStub) ~= "table" or type(libStub.GetLibrary) ~= "function" then
        return nil
    end
    local ok, library = pcall(
        libStub.GetLibrary,
        libStub,
        "LibForeverInspector",
        true
    )
    if ok and type(library) == "table" then
        return library
    end
    ok, library = pcall(
        libStub.GetLibrary,
        libStub,
        "LibClassicInspector",
        true
    )
    if ok and type(library) == "table" then

    end
    return nil
end
